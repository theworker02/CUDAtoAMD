cmake_minimum_required(VERSION 3.20)
project(compatcuda VERSION 1.3.0 LANGUAGES C CXX)

option(COMPATCUDA_ENABLE_HIP "Build the HIP-backed runtime implementation" OFF)
option(COMPATCUDA_BUILD_TESTS "Build ABI smoke tests" OFF)
option(COMPATCUDA_ENABLE_FFT "Enable optional hipFFT adapter when installed" ON)
set(COMPATCUDA_TEST_ARCH "gfx1101" CACHE STRING "GPU target for the native module correctness fixture")

add_library(compatcuda SHARED src/runtime.cpp src/amd_fft.cpp src/amd_graph.cpp src/cuda_libraries.cpp)
add_library(compatcuda::runtime ALIAS compatcuda)
target_compile_features(compatcuda PUBLIC cxx_std_17)
target_include_directories(compatcuda PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_definitions(compatcuda PRIVATE COMPATCUDA_BUILDING_LIBRARY)
set_target_properties(compatcuda PROPERTIES OUTPUT_NAME compatcuda WINDOWS_EXPORT_ALL_SYMBOLS ON)
include(GNUInstallDirs)
install(TARGETS compatcuda EXPORT compatcuda-targets
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE LEGAL.md SECURITY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/compatcuda)

if(COMPATCUDA_ENABLE_HIP)
  target_sources(compatcuda PRIVATE src/amd_runtime.cpp src/amd_blas.cpp)
  find_package(hip REQUIRED)
  find_package(hipblas REQUIRED)
  if(COMPATCUDA_ENABLE_FFT)
    find_package(hipfft QUIET HINTS "${hip_DIR}/../hipfft")
    if(hipfft_FOUND)
      target_link_libraries(compatcuda PRIVATE hip::hipfft)
      target_compile_definitions(compatcuda PRIVATE COMPATCUDA_ENABLE_HIPFFT=1)
    else()
      message(STATUS "hipFFT not found: FFT API will return backend-unavailable errors")
    endif()
  endif()
  target_link_libraries(compatcuda PRIVATE hip::host)
  target_link_libraries(compatcuda PRIVATE roc::hipblas)
  target_compile_definitions(compatcuda PRIVATE COMPATCUDA_ENABLE_HIP=1)
  get_filename_component(COMPATCUDA_ROCM_ROOT "${hip_DIR}/../../.." ABSOLUTE)
  set(COMPATCUDA_ROCM_BIN "${COMPATCUDA_ROCM_ROOT}/bin")
else()
  target_sources(compatcuda PRIVATE src/amd_disabled.cpp)
endif()

if(COMPATCUDA_ENABLE_HIP)
  find_program(COMPATCUDA_HIPCC NAMES hipcc.exe hipcc HINTS "${COMPATCUDA_ROCM_BIN}" REQUIRED)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/neural.hsaco"
    COMMAND "${COMPATCUDA_HIPCC}" --genco --no-gpu-bundle-output --offload-arch=${COMPATCUDA_TEST_ARCH}
      "${CMAKE_CURRENT_SOURCE_DIR}/kernels/neural.hip" -o "${CMAKE_CURRENT_BINARY_DIR}/neural.hsaco"
    DEPENDS kernels/neural.hip VERBATIM)
  add_custom_target(compatcuda_neural ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/neural.hsaco")
endif()

if(COMPATCUDA_BUILD_TESTS)
  enable_testing()
  add_executable(compatcuda_abi_smoke tests/native/abi_smoke.cpp)
  target_link_libraries(compatcuda_abi_smoke PRIVATE compatcuda)
  if(COMPATCUDA_ENABLE_HIP AND WIN32)
    add_test(NAME compatcuda_abi_smoke COMMAND ${CMAKE_COMMAND} -E env "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_abi_smoke>)
  else()
    add_test(NAME compatcuda_abi_smoke COMMAND compatcuda_abi_smoke)
  endif()
  add_executable(compatcuda_header_smoke tests/native/header_smoke.cpp)
  target_link_libraries(compatcuda_header_smoke PRIVATE compatcuda)
  if(COMPATCUDA_ENABLE_HIP AND WIN32)
    add_test(NAME compatcuda_header_smoke COMMAND ${CMAKE_COMMAND} -E env "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_header_smoke>)
  else()
    add_test(NAME compatcuda_header_smoke COMMAND compatcuda_header_smoke)
  endif()
  add_executable(compatcuda_c_header_smoke tests/native/c_header_smoke.c)
  target_link_libraries(compatcuda_c_header_smoke PRIVATE compatcuda)
  if(COMPATCUDA_ENABLE_HIP AND WIN32)
    add_test(NAME compatcuda_c_header_smoke COMMAND ${CMAKE_COMMAND} -E env "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_c_header_smoke>)
  else()
    add_test(NAME compatcuda_c_header_smoke COMMAND compatcuda_c_header_smoke)
  endif()
  if(COMPATCUDA_ENABLE_HIP)
    find_program(COMPATCUDA_HIPCC NAMES hipcc.exe hipcc HINTS "${COMPATCUDA_ROCM_BIN}" REQUIRED)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco"
      COMMAND "${COMPATCUDA_HIPCC}" --genco --no-gpu-bundle-output --offload-arch=${COMPATCUDA_TEST_ARCH}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/native/vector_add.hip" -o "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco"
      DEPENDS tests/native/vector_add.hip VERBATIM)
    add_custom_target(compatcuda_kernel_fixture DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco")
    add_executable(compatcuda_module_correctness tests/native/module_correctness.cpp)
    target_compile_definitions(compatcuda_module_correctness PRIVATE COMPATCUDA_TEST_ARCH="${COMPATCUDA_TEST_ARCH}")
    target_link_libraries(compatcuda_module_correctness PRIVATE compatcuda)
    add_dependencies(compatcuda_module_correctness compatcuda_kernel_fixture)
    if(WIN32)
      add_test(NAME compatcuda_module_correctness COMMAND ${CMAKE_COMMAND} -E env
        "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_module_correctness> "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco")
    else()
      add_test(NAME compatcuda_module_correctness COMMAND compatcuda_module_correctness "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco")
    endif()
    set_tests_properties(compatcuda_module_correctness PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
    if(hipfft_FOUND AND COMPATCUDA_ENABLE_FFT)
      add_executable(compatcuda_contract tests/native/cuda_contract.cpp)
      target_link_libraries(compatcuda_contract PRIVATE compatcuda)
      add_dependencies(compatcuda_contract compatcuda_kernel_fixture)
      if(WIN32)
        add_test(NAME compatcuda_contract COMMAND ${CMAKE_COMMAND} -E env "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_contract> "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco")
      else()
        add_test(NAME compatcuda_contract COMMAND compatcuda_contract "${CMAKE_CURRENT_BINARY_DIR}/vector_add.hsaco")
      endif()
      set_tests_properties(compatcuda_contract PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
    endif()
    add_executable(compatcuda_hip_core tests/native/hip_core.cpp)
    target_link_libraries(compatcuda_hip_core PRIVATE compatcuda)
    if(WIN32)
      add_test(NAME compatcuda_hip_core COMMAND ${CMAKE_COMMAND} -E env "PATH=${COMPATCUDA_ROCM_BIN};$ENV{PATH}" $<TARGET_FILE:compatcuda_hip_core>)
    else()
      add_test(NAME compatcuda_hip_core COMMAND compatcuda_hip_core)
    endif()
    set_tests_properties(compatcuda_hip_core PROPERTIES SKIP_RETURN_CODE 77)
  else()
    add_executable(compatcuda_disabled tests/native/disabled.cpp)
    target_link_libraries(compatcuda_disabled PRIVATE compatcuda)
    add_test(NAME compatcuda_disabled COMMAND compatcuda_disabled)
  endif()
endif()
